-- Family Chores 012: avatar voice learning profiles, and the optional AI voice (off by default).
-- Run after 001–011. Safe to re-run. Ends with its own tests (test data removed).

alter table public.settings add column if not exists ai_voice boolean not null default false;
alter table public.settings add column if not exists ai_daily_lines smallint not null default 20;      -- per person per day
alter table public.settings add column if not exists ai_monthly_cap_cents integer not null default 300; -- family cap, estimated cost
do $$ begin
  if not exists (select 1 from pg_constraint where conname = 'settings_ai_limits') then
    alter table public.settings add constraint settings_ai_limits check (ai_daily_lines between 0 and 200 and ai_monthly_cap_cents between 0 and 5000); end if;
end $$;

-- What the built-in voice engine has learned about each person (style stats, best hours, recent lines). Private to that person.
create table if not exists public.voice_profiles (
  member_id uuid primary key references public.members(id) on delete cascade,
  family_id uuid not null references public.families(id) on delete cascade,
  profile jsonb not null default '{}',
  updated_at timestamptz not null default now(),
  check (pg_column_size(profile) < 20000)
);
alter table public.voice_profiles enable row level security;
revoke all on public.voice_profiles from anon, authenticated;
grant select, insert, update on public.voice_profiles to authenticated;
drop policy if exists vp_all on public.voice_profiles;
create policy vp_all on public.voice_profiles for all to authenticated using (member_id = public.my_member())
  with check (member_id = public.my_member() and family_id = public.my_family());

-- AI usage ledger (written by the avatar-voice server function only); parents can see totals.
create table if not exists public.ai_usage (
  id bigint generated always as identity primary key,
  family_id uuid not null references public.families(id) on delete cascade,
  member_id uuid references public.members(id) on delete set null,
  kind text not null default 'voice',
  input_tokens integer not null default 0,
  output_tokens integer not null default 0,
  est_cost_micro integer not null default 0,        -- millionths of a dollar
  at timestamptz not null default now()
);
alter table public.ai_usage enable row level security;
revoke all on public.ai_usage from anon, authenticated;
grant select on public.ai_usage to authenticated;
drop policy if exists ai_read on public.ai_usage;
create policy ai_read on public.ai_usage for select to authenticated using (public.is_parent_of(family_id));

-- Server-side gate used by the avatar-voice function (service role only): is AI on, and is this person under the limits?
create or replace function public.ai_voice_allowed(member uuid) returns boolean
language sql stable security definer set search_path = '' as $$
  select s.ai_voice
     and (select count(*) from public.ai_usage u where u.member_id = member and u.at > now() - interval '1 day') < s.ai_daily_lines
     and (select coalesce(sum(u.est_cost_micro), 0) from public.ai_usage u where u.family_id = m.family_id and u.at > date_trunc('month', now())) < s.ai_monthly_cap_cents * 10000
  from public.members m join public.settings s on s.family_id = m.family_id where m.id = member $$;
revoke all on function public.ai_voice_allowed(uuid) from public, anon, authenticated;
grant execute on function public.ai_voice_allowed(uuid) to service_role;
grant insert on public.ai_usage to service_role;

-- only admins switch the AI voice on/off or change its limits
create or replace function public.guard_ai_settings() returns trigger
language plpgsql security definer set search_path = '' as $$
begin
  if (new.ai_voice, new.ai_daily_lines, new.ai_monthly_cap_cents) is distinct from (old.ai_voice, old.ai_daily_lines, old.ai_monthly_cap_cents)
     and auth.uid() is not null and not public.is_admin_of(new.family_id) then raise exception 'Only an admin can change the AI voice settings'; end if;
  return new;
end $$;
drop trigger if exists settings_ai_guard on public.settings;
create trigger settings_ai_guard before update on public.settings for each row execute function public.guard_ai_settings();
