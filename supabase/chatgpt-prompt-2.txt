-- Parent sets or resets a child's username + password (6–40 characters).
create or replace function public.set_kid_login(member uuid, uname text, pass text) returns void
language plpgsql security definer set search_path = '' as $$
declare m public.members;
begin
  select * into m from public.members where id = member;
  if m.id is null or not public.is_parent_of(m.family_id) then raise exception 'Only a parent can do this'; end if;
  if m.role <> 'child' then raise exception 'Parents sign in with email'; end if;
  if char_length(pass) < 6 or char_length(pass) > 40 then raise exception 'Password must be 6 to 40 characters'; end if;
  update public.members set username = lower(trim(uname)) where id = member;
  insert into public.member_login(member_id, pass_hash) values (member, extensions.crypt(pass, extensions.gen_salt('bf', 10)))
    on conflict (member_id) do update set pass_hash = excluded.pass_hash, failed_attempts = 0, locked_until = null, updated_at = now();
  perform public.log(m.family_id, 'kid_login_set', jsonb_build_object('member', member));
end $$;

-- Parent unlocks a child after too many wrong passwords.
create or replace function public.unlock_kid(member uuid) returns void
language plpgsql security definer set search_path = '' as $$
declare fam uuid; begin
  select family_id into fam from public.members where id = member;
  if fam is null or not public.is_parent_of(fam) then raise exception 'Only a parent can do this'; end if;
  update public.member_login set failed_attempts = 0, locked_until = null where member_id = member;
  perform public.log(fam, 'kid_unlocked', jsonb_build_object('member', member));
end $$;

-- Called ONLY by the kid-login server function (service role). Checks password, lockout and device.
create or replace function public.kid_auth(uname text, pass text, device text)
returns table(member_id uuid, user_id uuid, family_id uuid, error text)
language plpgsql security definer set search_path = '' as $$
#variable_conflict use_column
declare m public.members; l public.member_login;
begin
  select * into m from public.members where username = lower(trim(uname)) and role = 'child' and active;
  if m.id is null then return query select null::uuid, null::uuid, null::uuid, 'Wrong username or password'::text; return; end if;
  select * into l from public.member_login where member_login.member_id = m.id;
  if l.member_id is null then return query select null::uuid, null::uuid, null::uuid, 'Wrong username or password'::text; return; end if;
  if l.locked_until is not null and l.locked_until > now() then
    return query select null::uuid, null::uuid, null::uuid, 'Too many tries. Ask a parent, or wait 15 minutes.'::text; return; end if;
  if extensions.crypt(pass, l.pass_hash) <> l.pass_hash then
    update public.member_login set failed_attempts = failed_attempts + 1,
      locked_until = case when failed_attempts + 1 >= 5 then now() + interval '15 minutes' else null end
      where member_login.member_id = m.id;
    if l.failed_attempts + 1 >= 5 then
      insert into public.audit_log(family_id, actor, action, detail) values (m.family_id, m.id, 'kid_locked', null);
    end if;
    return query select null::uuid, null::uuid, null::uuid, 'Wrong username or password'::text; return;
  end if;
  if m.device_restricted and not exists (select 1 from public.member_devices d where d.member_id = m.id and d.device_id = device) then
    return query select null::uuid, null::uuid, null::uuid, 'This device isn''t allowed for your account. Ask a parent.'::text; return;
  end if;
  update public.member_login set failed_attempts = 0, locked_until = null where member_login.member_id = m.id;
  return query select m.id, m.user_id, m.family_id, null::text;
end $$;

-- Called ONLY by the kid-login function after it creates the child's hidden sign-in account.
create or replace function public.link_kid_user(member uuid, uid uuid) returns void
language sql security definer set search_path = '' as $$
  update public.members set user_id = uid where id = member and role = 'child' and user_id is null $$;

revoke all on function public.set_kid_login(uuid,text,text), public.unlock_kid(uuid), public.kid_auth(text,text,text), public.link_kid_user(uuid,uuid) from public, anon, authenticated;

select 'Kid logins installed' as result;

-- Family Chores: a parent picks up a chore/shift: for cash, as a penalty, or as a demonstration.
-- Run after 001 (and 003). Safe to re-run. Ends with its own tests (test data removed).

create table if not exists public.takeovers (
  id uuid primary key default gen_random_uuid(),
  family_id uuid not null references public.families(id) on delete cascade,
  job_id uuid not null references public.jobs(id) on delete cascade,
  occ date not null,
  child_member uuid references public.members(id) on delete set null,   -- null = an unclaimed/open shift
  parent_member uuid not null references public.members(id) on delete cascade,
  reason text not null check (reason in ('cash','penalty','demo')),
  penalty_id uuid references public.penalties(id) on delete set null,
  note text check (char_length(note) <= 120),
  created_at timestamptz not null default now(),
  unique (job_id, occ)
);
alter table public.takeovers enable row level security;
revoke all on public.takeovers from anon, authenticated;
grant select on public.takeovers to authenticated;
drop policy if exists take_read on public.takeovers;
create policy take_read on public.takeovers for select to authenticated using (public.is_member_of(family_id));

-- cash: chore moves to the parent, who earns its pay when scored.
-- penalty: chore moves to the parent; the child gets a penalty (default = the job's pay).
-- demo: parent shows how; the chore stays the child's; no money changes.
create or replace function public.parent_takeover(job uuid, day date, child uuid, why text, penalty_cents integer default null, note text default null)
returns uuid language plpgsql security definer set search_path = '' as $$
declare j public.jobs; me uuid := public.my_member(); t uuid; pen uuid; amt int;
begin
  select * into j from public.jobs where id = job;
  if j.id is null or not public.is_parent_of(j.family_id) then raise exception 'Only a parent can do this'; end if;
  if why not in ('cash','penalty','demo') then raise exception 'Reason must be cash, penalty or demo'; end if;
  if child is not null and not exists (select 1 from public.members where id = child and family_id = j.family_id and role = 'child') then
    raise exception 'That child is not in this family'; end if;
  if why = 'penalty' then
    if child is null then raise exception 'A penalty needs a child'; end if;
    amt := coalesce(penalty_cents, j.pay_cents);
    if amt < 1 or amt > 100000 then raise exception 'Penalty must be between 0.01 and 1000'; end if;
    insert into public.penalties(family_id, member_id, amount_cents, reason, created_by)
      values (j.family_id, child, amt, left('Parent took over "' || j.name || '"' || coalesce(': ' || note, ''), 120), me)
      returning id into pen;
  end if;
  insert into public.takeovers(family_id, job_id, occ, child_member, parent_member, reason, penalty_id, note)
    values (j.family_id, job, day, child, me, why, pen, note)
    on conflict (job_id, occ) do update set child_member = excluded.child_member, parent_member = excluded.parent_member,
      reason = excluded.reason, penalty_id = excluded.penalty_id, note = excluded.note, created_at = now()
    returning id into t;
  perform public.log(j.family_id, 'takeover_' || why, jsonb_build_object('job', job, 'occ', day, 'child', child, 'penalty_cents', amt));
  return t;
end $$;

-- Undo a takeover (and its penalty).

// Supabase Edge Function: kid-login
// A child signs in with username + password (+ this device's id). No email is ever used or sent.
// Deploy with "Verify JWT" OFF (the child isn't signed in yet). Secrets come from the function's own
// environment (SUPABASE_URL, SUPABASE_SERVICE_ROLE_KEY); nothing secret is in this file.
// Flow: kid_auth() checks password/lockout/device → (first time) create a hidden auth user with a
// placeholder address on the reserved .invalid domain (never deliverable) → return a one-time sign-in
// token the app exchanges for a normal session.

const URL_ = Deno.env.get("SUPABASE_URL")!;
const SERVICE = Deno.env.get("SUPABASE_SERVICE_ROLE_KEY")!;
const ALLOWED_ORIGINS = ["https://lorneh12.github.io", "http://localhost:8080"];

const cors = (origin: string | null) => ({
  "Access-Control-Allow-Origin": origin && ALLOWED_ORIGINS.includes(origin) ? origin : ALLOWED_ORIGINS[0],
  "Access-Control-Allow-Headers": "content-type, apikey, authorization, x-client-info",
  "Access-Control-Allow-Methods": "POST, OPTIONS",
  "Vary": "Origin",
});
const admin = (path: string, init: RequestInit = {}) =>
  fetch(`${URL_}${path}`, { ...init, headers: { apikey: SERVICE, Authorization: `Bearer ${SERVICE}`, "Content-Type": "application/json", ...(init.headers || {}) } });

Deno.serve(async (req) => {
  const origin = req.headers.get("origin");
  const h = { ...cors(origin), "Content-Type": "application/json" };
  if (req.method === "OPTIONS") return new Response("ok", { headers: h });
  if (req.method !== "POST") return new Response(JSON.stringify({ error: "POST only" }), { status: 405, headers: h });
  let body: { username?: string; password?: string; device_id?: string };
  try { body = await req.json(); } catch { return new Response(JSON.stringify({ error: "Bad request" }), { status: 400, headers: h }); }
  const username = String(body.username || "").slice(0, 20), password = String(body.password || "").slice(0, 40), device = String(body.device_id || "").slice(0, 64);
  if (!username || !password) return new Response(JSON.stringify({ error: "Enter your username and password" }), { status: 400, headers: h });

  const r = await admin("/rest/v1/rpc/kid_auth", { method: "POST", body: JSON.stringify({ uname: username, pass: password, device }) });
  const rows = await r.json();
  const row = Array.isArray(rows) ? rows[0] : null;
  if (!row || row.error) return new Response(JSON.stringify({ error: row?.error || "Sign-in failed" }), { status: 401, headers: h });

  const email = `kid-${row.member_id}@family-chores.invalid`;
  let uid: string | null = row.user_id;
  if (!uid) {
    const cu = await admin("/auth/v1/admin/users", { method: "POST", body: JSON.stringify({
      email, email_confirm: true, password: crypto.randomUUID() + crypto.randomUUID(), user_metadata: { kind: "child" } }) });
    const u = await cu.json();
    if (!cu.ok || !u.id) return new Response(JSON.stringify({ error: "Could not create the sign-in" }), { status: 500, headers: h });
    uid = u.id;
    await admin("/rest/v1/rpc/link_kid_user", { method: "POST", body: JSON.stringify({ member: row.member_id, uid }) });
  }
  // One-time token (no email is sent by generate_link).
  const gl = await admin("/auth/v1/admin/generate_link", { method: "POST", body: JSON.stringify({ type: "magiclink", email }) });
  const link = await gl.json();
  const token_hash = link.hashed_token || link.properties?.hashed_token;
  if (!gl.ok || !token_hash) return new Response(JSON.stringify({ error: "Could not sign in" }), { status: 500, headers: h });
  return new Response(JSON.stringify({ token_hash }), { headers: h });
});
